cmake_minimum_required(VERSION 3.16)
project(pcl_grid_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Boost REQUIRED)
if(NOT PCL_SOURCE_DIR OR NOT PCL_BUILD_DIR)
  message(FATAL_ERROR "Set PCL_SOURCE_DIR and PCL_BUILD_DIR to the pinned source and build")
endif()
add_executable(pcl-grid-check grid-check.cpp)
foreach(module common features search kdtree octree filters sample_consensus)
  target_include_directories(pcl-grid-check PRIVATE "${PCL_SOURCE_DIR}/${module}/include")
endforeach()
target_include_directories(pcl-grid-check PRIVATE "${PCL_BUILD_DIR}/include" ${Boost_INCLUDE_DIRS})
target_link_directories(pcl-grid-check PRIVATE "${PCL_BUILD_DIR}/lib")
target_link_libraries(pcl-grid-check PRIVATE Eigen3::Eigen pcl_features pcl_search pcl_kdtree pcl_octree pcl_common)

add_executable(pcl-segmentation-check segmentation-check.cpp)
foreach(module common features search kdtree octree filters sample_consensus segmentation)
  target_include_directories(pcl-segmentation-check PRIVATE "${PCL_SOURCE_DIR}/${module}/include")
endforeach()
target_include_directories(pcl-segmentation-check PRIVATE "${PCL_BUILD_DIR}/include" ${Boost_INCLUDE_DIRS})
target_link_directories(pcl-segmentation-check PRIVATE "${PCL_BUILD_DIR}/lib")
target_link_libraries(pcl-segmentation-check PRIVATE Eigen3::Eigen pcl_filters pcl_sample_consensus pcl_search pcl_kdtree pcl_octree pcl_common)
enable_testing()
add_test(NAME grid-failures COMMAND ${CMAKE_COMMAND}
  -DCHECKER=$<TARGET_FILE:pcl-grid-check>
  -P ${CMAKE_CURRENT_SOURCE_DIR}/grid-failures.cmake)
add_test(NAME segmentation-contracts COMMAND pcl-segmentation-check)

add_executable(pcl-registration-check registration-check.cpp)
foreach(module common registration)
  target_include_directories(pcl-registration-check PRIVATE "${PCL_SOURCE_DIR}/${module}/include")
endforeach()
target_include_directories(pcl-registration-check PRIVATE "${PCL_BUILD_DIR}/include" ${Boost_INCLUDE_DIRS})
target_link_directories(pcl-registration-check PRIVATE "${PCL_BUILD_DIR}/lib")
target_link_libraries(pcl-registration-check PRIVATE Eigen3::Eigen pcl_common)
add_test(NAME registration-contracts COMMAND pcl-registration-check)
