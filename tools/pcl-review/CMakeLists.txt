cmake_minimum_required(VERSION 3.16)
project(pcl_grid_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Boost REQUIRED)
if(NOT PCL_SOURCE_DIR OR NOT PCL_BUILD_DIR)
  message(FATAL_ERROR "Set PCL_SOURCE_DIR and PCL_BUILD_DIR to the pinned source and build")
endif()
add_executable(pcl-grid-check grid-check.cpp)
foreach(module common features search kdtree octree filters sample_consensus)
  target_include_directories(pcl-grid-check PRIVATE "${PCL_SOURCE_DIR}/${module}/include")
endforeach()
target_include_directories(pcl-grid-check PRIVATE "${PCL_BUILD_DIR}/include" ${Boost_INCLUDE_DIRS})
target_link_directories(pcl-grid-check PRIVATE "${PCL_BUILD_DIR}/lib")
target_link_libraries(pcl-grid-check PRIVATE Eigen3::Eigen pcl_features pcl_search pcl_kdtree pcl_octree pcl_common)
